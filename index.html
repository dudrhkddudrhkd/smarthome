ESP32 마이크로컨트롤러에 업로드할 MicroPython 예제 코드입니다. UART BLE 통신 규격으로 문자열을 수신하여 GPIO 14(빨강), 27(초록), 12(파랑)번에 연결된 RGB LED 색상을 제어합니다.

# ESP32 MicroPython Nordic UART BLE Service Firmware
import bluetooth
import machine
import neopixel
import time

# BLE UART UUIDs (Nordic UART Service)
_UART_UUID = bluetooth.UUID("6e400001-b5a3-f393-e0a9-e50e24dcca9e")
_FLAG_WRITE = 0x0008
_FLAG_NOTIFY = 0x0010
_UART_TX = (bluetooth.UUID("6e400003-b5a3-f393-e0a9-e50e24dcca9e"), _FLAG_NOTIFY)
_UART_RX = (bluetooth.UUID("6e400002-b5a3-f393-e0a9-e50e24dcca9e"), _FLAG_WRITE)
_UART_SERVICE = (_UART_UUID, (_UART_TX, _UART_RX))

class BLESimplePeripheral:
    def __init__(self, ble, name="ESP_LED"):
        self._ble = ble
        self._ble.active(True)
        self._ble.irq(self._irq)
        ((self._handle_tx, self._handle_rx),) = self._ble.gatts_register_services((_UART_SERVICE,))
        self._connections = set()
        self._write_callback = None
        self._name = name
        self._advertise()

    def _irq(self, event, data):
        if event == 1: # _IRQ_CENTRAL_CONNECT
            conn_handle, _, _ = data
            self._connections.add(conn_handle)
            print("Connected")
        elif event == 2: # _IRQ_CENTRAL_DISCONNECT
            conn_handle, _, _ = data
            self._connections.remove(conn_handle)
            print("Disconnected")
            self._advertise()
        elif event == 3: # _IRQ_GATTS_WRITE
            conn_handle, value_handle = data
            if value_handle == self._handle_rx:
                value = self._ble.gatts_read(self._handle_rx)
                if self._write_callback:
                    self._write_callback(value)

    def _advertise(self, interval_us=500000):
        # Flags (LE General Discoverable Mode, BR/EDR Not Supported)
        payload = bytearray(b"\x02\x01\x06")
        
        # Local Name
        name_bytes = self._name.encode("utf-8")
        payload.append(len(name_bytes) + 1)
        payload.append(0x09)
        payload.extend(name_bytes)
        
        # 128-bit Service UUIDs (Complete List)
        uuid_bytes = bytes(_UART_UUID)
        payload.append(len(uuid_bytes) + 1)
        payload.append(0x07)
        payload.extend(uuid_bytes)
        
        self._ble.gap_advertise(interval_us, adv_data=payload)

    def on_write(self, callback):
        self._write_callback = callback

# RGB LED Configuration (Common Cathode)
pin_r = machine.Pin(14, machine.Pin.OUT)
pin_g = machine.Pin(27, machine.Pin.OUT)
pin_b = machine.Pin(12, machine.Pin.OUT)

def set_color(r, g, b):
    # Common Cathode (GND common): 1 turns ON, 0 turns OFF
    # If using Common Anode (VCC common), invert logic: pin_r.value(0 if r > 0 else 1)
    pin_r.value(1 if r > 0 else 0)
    pin_g.value(1 if g > 0 else 0)
    pin_b.value(1 if b > 0 else 0)

# BLE Data Listener Callback
def on_rx(data):
    # Strip whitespace & carriage returns
    msg = data.decode("utf-8").strip()
    print("Received Command:", msg)
    
    if msg == "1":
        set_color(255, 0, 0)      # 빨강 (Red)
    elif msg == "2":
        set_color(0, 255, 0)      # 초록 (Green)
    elif msg == "abc":
        set_color(0, 0, 255)      # 파랑 (Blue)
    elif msg == "4":
        set_color(255, 255, 0)    # 노랑 (Yellow)
    else:
        set_color(0, 0, 0)        # 소등 (Black)

# Initialize BLE Simple Peripheral
ble = bluetooth.BLE()
sp = BLESimplePeripheral(ble)
sp.on_write(on_rx)

print("ESP32 BLE UART Server is running...")
